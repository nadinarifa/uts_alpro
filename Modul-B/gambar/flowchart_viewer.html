totalRencana = 0, totalRealisasi = 0

realisasi < 0?"} H -- Ya --> G H -- Tidak --> I["totalKomponen = totalKomponen + 1
totalRencana = totalRencana + anggaranRencana
totalRealisasi = totalRealisasi + realisasi
selisih = realisasi - anggaranRencana"] I --> J{"realisasi < anggaranRencana?"} J -- Ya --> K["jumlahHemat = jumlahHemat + 1

totalKelebihan = totalKelebihan + selisih"] N --> O{"selisih > penyimpanganTerbesar?"} O -- Ya --> P["penyimpanganTerbesar = selisih
komponenTerbesar = namaKomponen"] O -- Tidak --> Q K --> Q[/"Input pilihan lanjut (y/t)"/] M --> Q P --> Q Q --> R{"Pilihan bukan y, Y, t, atau T?"} R -- Ya --> Q R -- Tidak --> S{"Pilihan y atau Y?"} S -- Ya --> C S -- Tidak --> T{"totalRencana > 0?"} T -- Ya --> U["persentasePenyerapan = (totalRealisasi / totalRencana) * 100"] T -- Tidak --> V["persentasePenyerapan = 0"] U --> W{"totalRealisasi < totalRencana?"} V --> W W -- Ya --> X["kondisiKegiatan = 'Hemat'"] W -- Tidak --> Y{"totalRealisasi == totalRencana?"} Y -- Ya --> Z["kondisiKegiatan = 'Sesuai Anggaran'"] Y -- Tidak --> AA["kondisiKegiatan = 'Melebihi Anggaran'"] X --> AB[/"Tampilkan Laporan Rekapitulasi:
